let lastDirectionX = 0; let lastDirectionY = 0; // 控制蛇的移动 document.getElementById('upBtn').addEventListener('click', () => { if (dy !== 10) { lastDirectionX = 0; lastDirectionY = -10; dx = 0; dy = -10; } }); document.getElementById('downBtn').addEventListener('click', () => { if (dy !== -10) { lastDirectionX = 0; lastDirectionY = 10; dx = 0; dy = 10; } }); document.getElementById('leftBtn').addEventListener('click', () => { if (dx !== 10) { lastDirectionX = -10; lastDirectionY = 0; dx = -10; dy = 0; } }); document.getElementById('rightBtn').addEventListener('click', () => { if (dx !== -10) { lastDirectionX = 10; lastDirectionY = 0; dx = 10; dy = 0; } }); // 更新蛇的位置，实现可穿过边界功能并降低速度 function update() { let head = { x: snake[0].x + dx / speedReducer, y: snake[0].y + dy / speedReducer }; if (head.x < 0) { head.x = canvas.width - 10; } else if (head.x >= canvas.width) { head.x = 0; } if (head.y < 0) { head.y = canvas.height - 10; } else if (head.y >= canvas.height) { head.y = 0; } snake.unshift(head); // 调整食物判断逻辑，考虑蛇移动距离变小的情况 if (Math.abs(head.x - food.x) < lastDirectionX / speedReducer && Math.abs(head.y - food.y) < lastDirectionY / speedReducer) { food = generateFood(); } else { snake.pop(); } }
